To add money (e.g., your Bitcoin Miner):

C#
// Adds 0.001 BTC and automatically updates the UI
ResourceManager.Instance.AddBitcoin(0.001f);


To buy something (e.g., your App Store Download Button):

C#
// Tries to spend $15. If they have enough, it deducts it and runs the code inside.
if (ResourceManager.Instance.SpendUSD(15.00f)) 
{
    IconManager.Instance.InstallApp(appToDownload);
}
else
{
    Debug.Log("Player is too broke to buy this app.");
}

Internet speed:
// Upgrade speed
BrowserManager.Instance.UpgradeInternetSpeed(0.5f);

// Displays the current speed dynamically on your speed test webpage text component
float currentMbps = BrowserManager.Instance.GetSpeedScoreMBps();
speedTestText.text = $"Current Connection: {currentMbps} Mbps";

Upgrades:
Upgrades require a scriptableobject reference.
For app/UI unlocks, UnlockManager gameobject in heirarchy is required to be setup.

Notifications:
if (NotificationManager.Instance != null)
{
    NotificationManager.Instance.SendNotification("System Alert", "Your internet speed was upgraded!");
}
